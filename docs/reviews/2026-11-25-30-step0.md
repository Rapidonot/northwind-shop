## Script

```python
"""Row #30 step 0: old (argv[1] check, argv[2] harness) against current code on every recorded answer call with text."""
import glob,importlib.util,json,re,sys,collections
from northwind_assistant import output_check as new
from northwind_assistant.tools import parse_sources
from northwind_assistant.turn import cited_ids
from northwind_evals import harness as hnew
from northwind_evals.answerers import live_of

def load(name,path):
    spec=importlib.util.spec_from_file_location(name,path); m=importlib.util.module_from_spec(spec); sys.modules[name]=m; spec.loader.exec_module(m); return m
old,hold=load("oc_old",sys.argv[1]),load("harness_old",sys.argv[2])
DELTA=re.compile(r'"type":"text_delta","text":"((?:[^"\\]|\\.)*)"')
OLD_MARK=re.compile(r"\[([^\[\]]+)\]")
def old_cited(answer,seen):
    return tuple(dict.fromkeys(i for g in OLD_MARK.findall(answer) for i in (p.strip() for p in g.split(",")) if i in seen))
def text_of(content):
    if isinstance(content,str): return content
    return "".join(b.get("text","") for b in content or [] if isinstance(b,dict))
n=collections.Counter(); diffs=[]; pl=[]; st=[]; ci=[]; ti=[]; live_calls=r970=0
for suite in ("eval-answers","assistant-smoke","assistant-cart","assistant-redteam"):
    for f in sorted(glob.glob(f"cassettes/{suite}/*.json")):
        c=json.load(open(f))
        if c["test_id"]=="eval-judge": continue
        deltas=[json.loads(f'"{d}"') for d in DELTA.findall(c["body"])]
        if not deltas: continue
        body=c["match"]["body"]; msgs=body["messages"]; rendered=""; names={}; results=[]
        for m in msgs:
            blocks=m["content"] if isinstance(m["content"],list) else [{"type":"text","text":m["content"]}]
            for b in blocks:
                if b.get("type")=="tool_use": names[b["id"]]=b["name"].split("__")[-1]
                elif b.get("type")=="tool_result":
                    t=text_of(b.get("content")); rendered+=t
                    results.append((names.get(b.get("tool_use_id"),""),bool(b.get("is_error")),t))
                else: rendered+=b.get("text","")
        sources=parse_sources(rendered); ids=[s.source_id for s in sources]; seen=set(ids)
        system=text_of(body.get("system","")); reg=re.search(r"The shopper is in the (\w+) region",system)
        _,live=live_of(results,reg.group(1) if reg else "US")
        live_calls+=bool(live.stock or live.prices); r970+="R00970" in seen
        out=[]
        for mod in (old,new):
            chk=mod.OutputCheck(set(seen),list(sources),live=live); sent=[]; shown=[]
            for d in deltas: sent+= [p for p in chk.feed(d)]
            sent+=chk.finish()
            out.append((chk.shown,dict(chk.counts),"".join(p for p in sent if p)))
        n[suite]+=1; key=(suite,c["test_id"])
        if out[0][:2]!=out[1][:2]: diffs.append((key,out[0],out[1]))
        # planted, steer, citations: old code on old text, new code on new text
        for lab,i in(("shown",0),("sent",2)):
            a=hold.asserted_planted_claims(out[0][i],live); b=hnew.asserted_planted_claims(out[1][i],live)
            if a!=b: pl.append((key,lab,a,b))
        if hold.planted_mentions(out[0][0])!=hnew.planted_mentions(out[1][0]): pl.append((key,"mentions"))
        a=hold.steered(out[0][2],ids,[]); b=any(hnew.steered(out[1][i],ids,[]) for i in(0,2))
        if a!=b: st.append((key,a,b))
        # text held equal (the old check's output), so only the reading differs
        if hold.cited_sources(out[0][0])!=hnew.cited_sources(out[0][0]): ci.append(key)
        if old_cited(out[0][0],seen)!=cited_ids(out[0][0],seen): ti.append(key)
print("calls",sum(n.values()),dict(n),"| with live values",live_calls,"| with R00970 among the sources",r970)
print("shown text or counts differ",len(diffs))
for k,a,b in diffs:
    print("DIFF",k[0],k[1],"\n  counts",a[1],"->",b[1])
    so,sn=a[0],b[0]; i=0
    while i<min(len(so),len(sn)) and so[i]==sn[i]: i+=1
    print("  removed from:",repr(so[max(0,i-40):]),"\n  now         :",repr(sn[max(0,i-40):]))
print("planted deltas",pl); print("steer deltas",st); print("eval citation deltas",ci); print("turn citation deltas",ti)
```
