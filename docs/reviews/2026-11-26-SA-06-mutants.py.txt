def run(cmd, cwd=None):
    p = subprocess.run(cmd, cwd=cwd, capture_output=True, text=True, timeout=600)
    return p.returncode, (p.stdout + p.stderr)


def summary_line(out):
    for line in reversed(out.splitlines()):
        s = line.strip()
        if any(w in s for w in (" passed", " failed", " error", "Tests ")):
            return s
    return out.strip().splitlines()[-1] if out.strip() else "(no output)"


def main():
    code, status = run(["git", "status", "--porcelain", "--untracked-files=no"])
    if status.strip():
        sys.exit("tree not clean: stop")
    head = run(["git", "rev-parse", "--short", "HEAD"])[1].strip()
    print(f"SA-06 mutants at {head} (tests unchanged; each mutant restored with git checkout)")
    caught = 0
    for mid, finding, path, old, new, kind, tests in MUTANTS:
        if old is None:
            run(["git", "checkout", new, "--", path])
        else:
            text = open(path, encoding="utf-8").read()
            if text.count(old) != 1:
                print(f"{mid} {finding}: ANCHOR NOT FOUND ({text.count(old)}) - not run")
                continue
            open(path, "w", encoding="utf-8").write(text.replace(old, new))
        try:
            if kind == "py":
                code, out = run(["uv", "run", "--quiet", "pytest", *tests, "-q", "-p", "no:cacheprovider"])
            else:
                vfile, pattern = tests
                code, out = run(["corepack", "pnpm", "exec", "vitest", "run", vfile, "-t", pattern],
                                cwd="apps/portal")
        finally:
            run(["git", "checkout", "HEAD", "--", path])
        verdict = "CAUGHT" if code != 0 else "SURVIVED"
        caught += code != 0
        print(f"{mid} {finding}: {verdict} - {summary_line(out)}")
    left = run(["git", "status", "--porcelain", "--untracked-files=no"])[1].strip()
    print(f"{caught} of {len(MUTANTS)} caught; tree after: {'clean' if not left else left}")


if __name__ == "__main__":
    main()
