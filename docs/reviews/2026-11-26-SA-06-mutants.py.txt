import json
import os
import re
import tempfile

GIT_ERRORS = ("Operation not permitted", "could not lock config file", "unable to unlink",
              "unable to create file", "cannot create directory")


def run(cmd, cwd=None):
    p = subprocess.run(cmd, cwd=cwd, capture_output=True, text=True, timeout=600)
    return p.returncode, (p.stdout + p.stderr)


def git(*args):
    code, out = run(["git", *args])
    if code != 0 or any(e in out for e in GIT_ERRORS):
        sys.exit(f"git {' '.join(args)} failed ({code}): {out.strip()[:300]}\nSTOP: check the tree by hand")
    return out


def write(path, text):
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(text)


def restore(path):
    write(path, git("show", f"HEAD:{path}"))
    code, _ = run(["git", "diff", "--quiet", "HEAD", "--", path])
    if code != 0:
        sys.exit(f"{path} is not HEAD's blob after the restore\nSTOP: check the tree by hand")


def pytest_verdict(tests):
    code, out = run(["uv", "run", "--quiet", "--no-sync", "pytest", *tests, "-q", "-rfEA",
                     "-p", "no:cacheprovider"])
    failed = re.findall(r"^FAILED (\S+)", out, re.M)
    passed = re.findall(r"^PASSED (\S+)", out, re.M)
    hit = all(any(f.startswith(t) for f in failed) for t in tests)
    verdict = "CAUGHT" if failed and hit else ("SURVIVED" if code == 0 else "ERROR")
    return verdict, f"{len(failed)} failed, {len(passed)} passed", failed, passed, out


def vitest_verdict(spec):
    vfile, pattern = spec
    fd, report = tempfile.mkstemp(suffix=".json")
    os.close(fd)
    code, out = run(["corepack", "pnpm", "exec", "vitest", "run", vfile, "-t", pattern,
                     "--reporter=json", f"--outputFile={report}"], cwd="apps/portal")
    try:
        data = json.load(open(report, encoding="utf-8"))
    except (OSError, ValueError):
        return "ERROR", "no JSON report", [], [], out
    finally:
        os.unlink(report)
    results = [a for f in data.get("testResults", []) for a in f.get("assertionResults", [])]
    named = [a for a in results if pattern in a.get("fullName", "")]
    failed = [a["fullName"] for a in named if a.get("status") == "failed"]
    passed = [a["fullName"] for a in named if a.get("status") == "passed"]
    errors = data.get("numFailedTestSuites", 0) or bool(data.get("unhandledErrors"))
    verdict = "CAUGHT" if failed else ("SURVIVED" if code == 0 else "ERROR")
    note = f"{len(failed)} named failed on assertion, {len(passed)} named passed"
    if errors:
        note += "; the run also had an unhandled error or failed suite"
    return verdict, note, failed, passed, out


def main():
    if git("status", "--porcelain", "--untracked-files=no").strip():
        sys.exit("tree not clean: stop")
    head = git("rev-parse", "--short", "HEAD").strip()
    print(f"SA-06 mutants v2 at {head} (tests unchanged; each file restored from HEAD and checked)")
    bad = 0
    for mid, finding, path, old, new, kind, tests in MUTANTS:
        if old is None:
            write(path, git("show", f"{new}:{path}"))  # the working tree only, never the index
        else:
            text = open(path, encoding="utf-8").read()
            if text.count(old) != 1:
                print(f"{mid} {finding}: ERROR - anchor found {text.count(old)} times, not run")
                bad += 1
                continue
            write(path, text.replace(old, new))
        try:
            verdict, note, failed, passed, _out = (
                pytest_verdict(tests) if kind == "py" else vitest_verdict(tests)
            )
        finally:
            restore(path)
        bad += verdict != "CAUGHT"
        print(f"{mid} {finding}: {verdict} - {note}")
        for name in passed:
            print(f"      passed: {name}")
    left = git("status", "--porcelain", "--untracked-files=no").strip()
    print(f"{len(MUTANTS) - bad} of {len(MUTANTS)} caught; tree after: {'clean' if not left else left}")
    if bad or left:
        sys.exit(1)


if __name__ == "__main__":
    main()
