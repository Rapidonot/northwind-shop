def run(tests):
    out = subprocess.run([PY, "-m", "pytest", "-p", "no:cacheprovider", "-q", "-ra", "--rootdir", str(ROOT), "-c", str(ROOT / "pyproject.toml"), *[str(ROOT / t) for t in tests]], cwd=ROOT, env=ENV, capture_output=True, text=True).stdout
    failed = sorted({re.sub(r"\[.*", "", m).split("::")[-1] for m in re.findall(r"^(?:FAILED|ERROR) (\S+)", out, re.M)})
    tail = [line for line in out.splitlines() if re.search(r"\d+ (passed|failed|error)", line)]
    return failed, (tail or [out[-300:]])[-1]
