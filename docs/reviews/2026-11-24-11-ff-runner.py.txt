def git(*a): return subprocess.run(['git',*a],capture_output=True,text=True)
assert not git('status','--porcelain').stdout.strip(), 'tree not clean'
out=[]
for label, swaps, mutation, tests in CASES:
    paths=[p for p,_ in swaps]
    for p,rev in swaps:
        r=git('show',f'{rev}:{p}'); assert r.returncode==0, (p,rev,r.stderr)
        open(p,'w').write(r.stdout)
    for p,old,new in (mutation or []):
        s=open(p).read(); assert old in s, (label, old); open(p,'w').write(s.replace(old,new,1)); paths.append(p)
    env=dict(os.environ, LLM_MODE='replay')
    r=subprocess.run(['uv','run','pytest','-q','-p','no:xdist',*tests],capture_output=True,text=True,env=env,timeout=900)
    lines=r.stdout.splitlines()
    summary=lines[-1] if lines else r.stderr[-200:]
    first_e=next((l for l in lines if l.startswith('E ')), '')
    git('checkout','HEAD','--',*paths)
    src=', '.join(f'{p}@{rev}' for p,rev in swaps) or 'mutation in ' + ', '.join(sorted({m[0] for m in mutation}))
    out.append(f"- **{label}** ({src}): `{summary}`" + (f"\n  `{first_e[:220]}`" if first_e else ''))
    print(out[-1], flush=True)
assert not git('status','--porcelain').stdout.strip(), 'tree not clean after'
