## Script

```python
"""Row #27 PR 2 step 0: every recorded answer call with text, through the importable tree's own check."""
import glob, json, re, sys
from pathlib import Path
import northwind_assistant.output_check as oc
from northwind_assistant.tools import parse_sources
from northwind_assistant.turn import cited_ids
from northwind_evals.answerers import live_of

if "--control" in sys.argv:  # drop one class: the comparison must see it
    kept = tuple(p for p in oc.PATTERNS if p.cls != "promotion")
    for m in (oc,):
        m.PATTERNS = kept
DELTA = re.compile(r'"type":"text_delta","text":"((?:[^"\\]|\\.)*)"')
SUITES = ("eval-answers", "assistant-smoke", "assistant-cart", "assistant-orders",
          "assistant-redteam", "assistant-intents", "a2a-05-partner")

def text_of(content):
    if isinstance(content, str):
        return content
    return "".join(b.get("text", "") for b in content or [] if isinstance(b, dict))

out = {"module": oc.__file__, "calls": {}}
for suite in SUITES:
    for f in sorted(glob.glob(f"cassettes/{suite}/*.json")):
        c = json.load(open(f))
        if c["test_id"] == "eval-judge":
            continue
        deltas = [json.loads(f'"{d}"') for d in DELTA.findall(c["body"])]
        if not deltas:
            continue
        body = c["match"]["body"]
        rendered, names, results = "", {}, []
        for m in body["messages"]:
            blocks = m["content"] if isinstance(m["content"], list) else [{"type": "text", "text": m["content"]}]
            for b in blocks:
                if b.get("type") == "tool_use":
                    names[b["id"]] = b["name"].split("__")[-1]
                elif b.get("type") == "tool_result":
                    t = text_of(b.get("content")); rendered += t
                    results.append((names.get(b.get("tool_use_id"), ""), bool(b.get("is_error")), t))
                else:
                    rendered += b.get("text", "")
        sources = parse_sources(rendered)
        seen = {s.source_id for s in sources}
        reg = re.search(r"The shopper is in the (\w+) region", text_of(body.get("system", "")))
        _, live = live_of(results, reg.group(1) if reg else "US")
        chk = oc.OutputCheck(set(seen), list(sources), live=live)
        sent = []
        for d in deltas:
            sent += chk.feed(d)
        sent += chk.finish()
        out["calls"][f"{suite}/{Path(f).name}"] = {
            "test": c["test_id"], "shown": chk.shown, "counts": dict(chk.counts),
            "sent": ["<reset>" if p is None else p for p in sent],
            "cited": list(cited_ids(chk.shown, seen)), "raw": "".join(deltas),
            "sources": len(sources), "live": bool(live.stock or live.prices),
        }
json.dump(out, open(sys.argv[1], "w"))
```
