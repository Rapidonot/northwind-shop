- **Classes:**

    and not in the raw text.
  - **self:** the assistant describing what it can or cannot do.
  - **other:** any other site or policy claim.
- **Asked:** the shopper's question asks about the assistant or the site ("can/could/will/do you", "are you",
  "website", "checkout", "account", "promo", "code"). This is a keyword heuristic and was not read by eye.
- **Target:** at most 5 model-made claims (self plus other). Reported, not gated.
- **Script:** at the end of this file. Run it as `uv run --offline python d8.py .llm-proxy-log/eval-last-run.json`.

## Script

```python
"""Row #17: D8 count by #32's method (docs/reviews/2026-11-18-32-d8.md), plus asked/unasked (addendum 1, item 3).
No LLM call. Claims: every claim the judge marked unsupported with kind site or policy, raw and after M1 (v6),
from the item's eval-judge cassette. Classes: check (restates a FIXED_SENTENCES value the check put there: in the
sent answer, not in the model's raw text; difflib >= 0.5), self (the assistant on what it can or cannot do),
other. Asked: the shopper's question asks about the assistant's abilities or the site."""

import difflib, glob, json, re, sys
from collections import Counter
from northwind_assistant.output_check import FIXED_SENTENCES
from northwind_evals.judge import parse_verdict

items = {}
for line in open("evals/fixtures/starter_set.jsonl"):
    if line.strip():
        it = json.loads(line)
        items[it["query"].strip()] = it
run = {r["item_id"]: r for r in json.load(open(sys.argv[1]))} if len(sys.argv) > 1 else {}
SELF = re.compile(
    r"^\s*(?:the |this )?(?:assistant|AI assistant|I|I'm|we|Northwind's assistant)\b|\b(?:assistant|I) (?:can|cannot|can't|could|will|has no|have no)\b",
    re.I,
)
ASKED = re.compile(
    r"\b(?:can|could|will|would|do) you\b|\bare you\b|\byour (?:tools|abilities)\b|\bwebsite\b|\bcheckout\b|\baccount\b|\bpromo|\bcode\b",
    re.I,
)
FIXED = set(FIXED_SENTENCES.values())
# The model's raw text: the item's eval-answerer cassettes (text deltas), as #32 read it.
RAW = {}
for f in sorted(glob.glob("cassettes/eval-answers/*.json")):
    d = json.load(open(f))
    if d["test_id"].startswith("eval-answerer:"):
        iid = d["test_id"].split(":", 1)[1].split("#")[0]
        deltas = re.findall(r'"type":"text_delta","text":"((?:[^"\\]|\\.)*)"', d["body"])
        RAW[iid] = RAW.get(iid, "") + "".join(json.loads(f'"{x}"') for x in deltas)
rows, mismatch = [], 0
for f in sorted(glob.glob("cassettes/eval-answers/*.json")):
    d = json.load(open(f))
    if d["test_id"] != "eval-judge":
        continue
    req = d["match"]["body"]["messages"][0]["content"]
    q = re.search(r"<question>(.*?)</question>", req, re.S).group(1).strip()
    ans = re.search(r"<answer>(.*?)</answer>", req, re.S).group(1)
    it = items.get(q)
    iid = it["id"] if it else "?"
    r = run.get(iid, {})
    if r and r.get("sent", "").strip() != ans.strip():
        mismatch += 1
    body = json.loads(d["body"])
    text = "".join(b.get("text", "") for b in body["content"])
    raw = json.loads(text)
    v = parse_verdict(text)
    final = {c: k for k, c in v.unsupported}
    for c in raw["claims"]:
        if c.get("supported") is True or c.get("kind") not in ("site", "policy"):
            continue
        claim = c["claim"]
        fk = final.get(claim, c["kind"])
        best = max(
            FIXED, key=lambda s: difflib.SequenceMatcher(None, claim.lower(), s.lower()).ratio()
        )
        ratio = difflib.SequenceMatcher(None, claim.lower(), best.lower()).ratio()
        # The shown answer is the judge's <answer>: eval-last-run's `sent` also holds text a Reset took back.
        inserted = best in ans and best not in RAW.get(iid, "")
        cls = "check" if ratio >= 0.5 and inserted else ("self" if SELF.search(claim) else "other")
        rows.append(
            dict(
                item=iid,
                segment=(it or {}).get("segment", "?"),
                raw=c["kind"],
                final=fk,
                cls=cls,
                asked=bool(ASKED.search(q)),
                claim=claim,
                question=q,
            )
        )


def table(sel, label):
    t = Counter(r["cls"] for r in sel)
    made = t["self"] + t["other"]
    unasked = sum(1 for r in sel if r["cls"] in ("self", "other") and not r["asked"])
    print(
        f"| {label} | {len(sel)} | {len({r['item'] for r in sel})} | {t['check']} | {t['self']} | {t['other']} | {made} | {unasked} |"
    )


print(f"judge answers that differ from eval-last-run `sent` (Reset text kept there): {mismatch}")
print("| | Claims | Items | check | self | other | Model-made | Model-made, unasked |")
print("|---|---|---|---|---|---|---|---|")
table(rows, "All (raw kinds)")
table([r for r in rows if r["final"] in ("site", "policy")], "All (after M1)")
table([r for r in rows if r["item"] <= "EV-102"], "EV-001 to 102")
table([r for r in rows if r["item"] > "EV-102"], "EV-103 to 308")
print(
    "\n| Item | Segment | Raw | After M1 | Class | Asked | Claim |\n|---|---|---|---|---|---|---|"
)
for r in sorted(rows, key=lambda r: r["item"]):
    print(
        f"| {r['item']} | {r['segment']} | {r['raw']} | {r['final']} | {r['cls']} | {'asked' if r['asked'] else 'unasked'} | {r['claim']} |"
    )
```
