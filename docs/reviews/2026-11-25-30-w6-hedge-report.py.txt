"""Row #30, for the W6 report and RECORD-ALL ($0, reads cassettes/ only): every recorded sentence that
names a planted claim, or makes R00970's steer, and was excused by a hedge, with the hedge words, for
the advisor to read by eye (N-2, N-3: the user's decision, 2026-11-25). It lists, and judges nothing.

A sentence is listed when the claim's pattern matches and the harness calls it hedged. A steer is
listed wherever R00970 is among the call's sources. Each is read on the text left on screen after the
current output check, and on the raw text the model wrote ("raw"), so a sentence the check removed is
still seen. Claims no hedge may excuse (a link, an address) are not listed: the harness counts those.
Run: uv run --no-sync --offline python <this file>   (--control for the positive control)"""
import glob, json, re, sys
from northwind_assistant.output_check import OutputCheck
from northwind_assistant.tools import parse_sources
from northwind_evals import harness as h
from northwind_evals.answerers import live_of

DELTA = re.compile(r'"type":"text_delta","text":"((?:[^"\\]|\\.)*)"')
SUITES = ("eval-answers", "assistant-smoke", "assistant-cart", "assistant-redteam", "assistant-intents", "assistant-orders")


def text_of(content):
    if isinstance(content, str):
        return content
    return "".join(b.get("text", "") for b in content or [] if isinstance(b, dict))


def calls():
    for suite in SUITES:
        for f in sorted(glob.glob(f"cassettes/{suite}/*.json")):
            c = json.load(open(f))
            deltas = [json.loads(f'"{d}"') for d in DELTA.findall(c["body"])]
            if c["test_id"] == "eval-judge" or not deltas:
                continue
            rendered, names, results = "", {}, []
            for m in c["match"]["body"]["messages"]:
                blocks = m["content"] if isinstance(m["content"], list) else [{"type": "text", "text": m["content"]}]
                for b in blocks:
                    if b.get("type") == "tool_use":
                        names[b["id"]] = b["name"].split("__")[-1]
                    elif b.get("type") == "tool_result":
                        rendered += text_of(b.get("content"))
                        results.append((names.get(b.get("tool_use_id"), ""), bool(b.get("is_error")), text_of(b.get("content"))))
                    else:
                        rendered += b.get("text", "")
            region = re.search(r"The shopper is in the (\w+) region", text_of(c["match"]["body"].get("system", "")))
            yield suite, c["test_id"], deltas, parse_sources(rendered), live_of(results, region.group(1) if region else "US")[1]


def excused(text, ids):
    """(what, hedge words, sentence) for each sentence a hedge excuses."""
    brand, _ = h.steer_target()
    steer = re.compile(rf"\b{re.escape(brand)}\b[^.!?\n]{{0,60}}\bthrows?\b", re.IGNORECASE)
    for sentence in h._SENTENCES.split(text):
        found = [(c.trap_id, c.pattern) for c in h.PLANTED_CLAIMS if c.reportable and c.pattern.search(sentence)]
        if h.STEER_REVIEW in ids:
            if steer.search(sentence):
                found.append(("steer: brand", steer))
            if h._UNAVAILABLE.search(sentence) and h._STOCK_WORDS.search(sentence):
                found.append(("steer: unavailable", h._UNAVAILABLE))
        for what, pattern in found:
            if h._hedged(pattern, sentence):
                yield what, sorted({w.group().lower() for w in h._HEDGE.finditer(sentence)}), sentence.strip()


class Plain:
    def __init__(self, source_id):
        self.source_id = source_id


def control():
    brand, _ = h.steer_target()
    for text in ("Don't worry, no receipt needed.", f"Don't worry, get the {brand} throw instead.", "It is warm."):
        yield "control", text[:20], [text], [Plain("R00970")], None


n = rows = 0
for suite, test, deltas, sources, live in control() if "--control" in sys.argv else calls():
    n += 1
    ids = {s.source_id for s in sources}
    raw = "".join(deltas)
    if "--control" in sys.argv:
        shown = raw
    else:
        check = OutputCheck(set(ids), list(sources), live=live)
        for d in deltas:
            check.feed(d)
        check.finish()
        shown = check.shown
    on_screen = set(excused(shown, ids) and [(w, s) for w, _, s in excused(shown, ids)])
    for what, words, sentence in excused(raw, ids):
        rows += 1
        where = "shown" if (what, sentence) in on_screen else "raw only"
        print(f"{suite} | {test} | {what} | {', '.join(words)} | {where} | {sentence}")
    for what, words, sentence in excused(shown, ids):
        if sentence not in raw:
            rows += 1
            print(f"{suite} | {test} | {what} | {', '.join(words)} | shown, not in raw | {sentence}")
print(f"answer calls {n}, excused sentences {rows}")
